"""Bounded, durable sequential research through the V6 application."""
import json, uuid
class Workflow:
    def __init__(self, app):
        self.app=app; self.db=app.core.db
        with self.db:
            self.db.execute('CREATE TABLE IF NOT EXISTS runs(id TEXT PRIMARY KEY, plan TEXT NOT NULL, status TEXT NOT NULL)')
            self.db.execute('CREATE TABLE IF NOT EXISTS steps(run TEXT, idx INTEGER, status TEXT, record TEXT, PRIMARY KEY(run,idx))')
    def run(self, questions, provider, max_steps=5):
        if type(max_steps) is not int or not 1<=max_steps<=20: raise ValueError('invalid step budget')
        if not isinstance(questions,list) or not 1<=len(questions)<=max_steps or any(not isinstance(q,str) or not q.strip() or len(q)>4000 for q in questions): raise ValueError('invalid plan')
        rid=str(uuid.uuid4())
        with self.db: self.db.execute('INSERT INTO runs VALUES (?,?,?)',(rid,json.dumps(questions),'running'))
        for i,q in enumerate(questions):
            with self.db: self.db.execute('INSERT INTO steps VALUES (?,?,?,?)',(rid,i,'running',None))
            try: record=self.app.investigate(q,provider)
            except Exception as exc:
                with self.db:
                    self.db.execute('UPDATE steps SET status=?,record=? WHERE run=? AND idx=?',('failed',type(exc).__name__,rid,i))
                    self.db.execute('UPDATE runs SET status=? WHERE id=?',('failed',rid))
                return self.status(rid)
            with self.db: self.db.execute('UPDATE steps SET status=?,record=? WHERE run=? AND idx=?',('complete',record['id'],rid,i))
        with self.db: self.db.execute('UPDATE runs SET status=? WHERE id=?',('complete',rid))
        return self.status(rid)
    def status(self,rid):
        row=self.db.execute('SELECT plan,status FROM runs WHERE id=?',(rid,)).fetchone()
        if row is None: raise KeyError(rid)
        return dict(id=rid,plan=json.loads(row[0]),status=row[1],steps=[dict(index=i,status=s,record=r) for i,s,r in self.db.execute('SELECT idx,status,record FROM steps WHERE run=? ORDER BY idx',(rid,))])
